<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8" />
  <meta name="viewport" content="width=device-width, initial-scale=1.0" />
  <title>Farm Simulator ROI - Fotra Micro-Farm</title>
  <script src="https://cdn.tailwindcss.com"></script>
  <script crossorigin src="https://unpkg.com/react@18/umd/react.production.min.js"></script>
  <script crossorigin src="https://unpkg.com/react-dom@18/umd/react-dom.production.min.js"></script>
  <script src="https://unpkg.com/@babel/standalone/babel.min.js"></script>
  <style>
    input[type=range] { accent-color: #059669; }
  </style>
</head>
<body class="bg-slate-50 text-slate-900 font-sans antialiased min-h-screen">
  <div id="root"></div>

  <script type="text/babel">
    const { useState, useMemo } = React;

    const CAPACITY_PRESETS = [
      { blocks: 40, label: '40 Blocks', desc: 'Spare room • ~5 hrs/wk' },
      { blocks: 100, label: '100 Blocks', desc: 'Restaurant supply • ~12 hrs/wk' },
      { blocks: 250, label: '250 Blocks', desc: 'Commercial facility • ~25 hrs/wk' }
    ];

    const INVESTMENT_PRESETS = [
      { label: 'Room Retrofit', value: 4500 },
      { label: 'Starter Standard', value: 7650 },
      { label: 'Boutique Commercial', value: 14648 },
      { label: 'Commercial Pod', value: 16248 }
    ];

    function App() {
      const [setupInvestment, setSetupInvestment] = useState(7650);
      const [monthlyBlocks, setMonthlyBlocks] = useState(40);
      const [outcomeTab, setOutcomeTab] = useState('timeline');

      const scaleProfile = useMemo(() => {
        if (monthlyBlocks <= 60) return { name: 'Starter Micro-Grower', commitment: '~5 hrs / week' };
        if (monthlyBlocks <= 150) return { name: 'Boutique Cultivator', commitment: '~12 hrs / week' };
        return { name: 'Commercial Pod Facility', commitment: '~25 hrs / week' };
      }, [monthlyBlocks]);

      const outcomes = useMemo(() => {
        const blocks = monthlyBlocks;
        const yieldPerBlock = 1.6;
        const priceKg = 20.0;
        const substrateCost = 12.5;
        const complimentaryBlocks = 10;

        const monthlyYieldKg = blocks * yieldPerBlock;
        const grossMonthlySales = monthlyYieldKg * priceKg;
        const monthlySubstrateExpense = blocks * substrateCost;
        const netMonthlyOperatingProfit = grossMonthlySales - monthlySubstrateExpense;
        const unitMarginPerBlock = 19.50;
        const marginPct = (unitMarginPerBlock / 32.00) * 100;

        const estimatedArk = blocks <= 40 ? 1650 : (blocks <= 100 ? 2648 : 3248);
        const arkCost = Math.min(setupInvestment, estimatedArk);
        const chamberCost = Math.max(0, setupInvestment - arkCost);

        const month1PaidBlocks = Math.max(0, blocks - complimentaryBlocks);
        const month1SubstrateOpEx = month1PaidBlocks * substrateCost;
        const month1Outflows = setupInvestment + month1SubstrateOpEx;

        const monthlySeries = [];
        let cumCash = 0;
        let breakevenMonth = null;

        for (let m = 1; m <= 12; m++) {
          const isM1 = m === 1;
          const mGross = grossMonthlySales;
          const mArk = isM1 ? -arkCost : 0;
          const mChamber = isM1 ? -chamberCost : 0;
          const mSubstrate = isM1 ? -month1SubstrateOpEx : -monthlySubstrateExpense;
          const mOutflows = mArk + mChamber + mSubstrate;
          const netCash = mGross + mOutflows;
          cumCash += netCash;

          if (cumCash >= 0 && breakevenMonth === null) breakevenMonth = m;

          monthlySeries.push({
            month: m,
            shortName: `M${m}`,
            inflow: mGross,
            substrateExpense: mSubstrate,
            netCash,
            cumCash
          });
        }

        const year1NetCash = cumCash;
        const paybackMonths = netMonthlyOperatingProfit > 0 
          ? Number((setupInvestment / netMonthlyOperatingProfit).toFixed(1)) 
          : 99;
        const year1Roi = setupInvestment > 0 ? (year1NetCash / setupInvestment) * 100 : 0;
        const annualNetProfit = netMonthlyOperatingProfit * 12;

        return {
          blocks,
          monthlyYieldKg,
          grossMonthlySales,
          netMonthlyOperatingProfit,
          arkCost,
          chamberCost,
          paybackMonths,
          breakevenMonth: breakevenMonth || Math.ceil(paybackMonths),
          year1NetCash,
          year1Roi,
          annualNetProfit,
          monthlySeries
        };
      }, [monthlyBlocks, setupInvestment]);

      return (
        <div className="pb-16 max-w-6xl mx-auto px-4 sm:px-6 pt-6 space-y-6">
          <header className="bg-white border border-slate-200 rounded-2xl p-4 flex flex-col sm:flex-row items-start sm:items-center justify-between gap-3 shadow-sm">
            <div>
              <div className="flex items-center gap-2">
                <h1 className="text-xl font-black text-slate-900">Farm Simulator ROI</h1>
                <span className="px-2 py-0.5 text-xs font-bold bg-emerald-100 text-emerald-800 rounded-full">Fotra Outgrower</span>
              </div>
              <p className="text-xs text-slate-500">Commercial Feasibility & Payback Calculator</p>
            </div>
            <button
              onClick={() => { setMonthlyBlocks(40); setSetupInvestment(7650); }}
              className="px-3 py-1.5 bg-slate-100 hover:bg-slate-200 text-slate-700 text-xs font-semibold rounded-xl"
            >
              Reset Defaults
            </button>
          </header>

          <section className="bg-white rounded-3xl p-6 sm:p-7 border border-slate-200 shadow-sm space-y-6">
            <div className="flex flex-col sm:flex-row justify-between pb-4 border-b border-slate-100 gap-2">
              <div>
                <h3 className="text-lg font-bold text-slate-900">Interactive Controls</h3>
                <p className="text-xs text-slate-500">Tune CapEx and monthly block capacity</p>
              </div>
              <span className="text-xs font-semibold px-2.5 py-1 rounded-lg bg-emerald-50 text-emerald-800 border border-emerald-200 self-start sm:self-auto">
                {scaleProfile.name} ({scaleProfile.commitment})
              </span>
            </div>

            <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
              {/* Dial 1 */}
              <div className="bg-slate-50 rounded-2xl p-5 border border-slate-200 space-y-4">
                <div className="flex justify-between items-center">
                  <div>
                    <span className="text-xs font-bold text-blue-600 uppercase">Dial 1</span>
                    <h4 className="text-sm font-bold text-slate-900">Initial Setup Investment (CapEx)</h4>
                  </div>
                  <div className="text-xl font-black text-blue-900 font-mono">${setupInvestment.toLocaleString()}</div>
                </div>
                <input
                  type="range" min="2000" max="22000" step="250"
                  value={setupInvestment} onChange={(e) => setSetupInvestment(Number(e.target.value))}
                  className="w-full h-2.5 bg-slate-200 rounded-lg cursor-pointer accent-blue-600"
                />
                <div className="flex justify-between text-xs font-mono text-slate-500">
                  <span>$2,000</span><span>$7,650</span><span>$14,648</span><span>$22,000</span>
                </div>
                <div className="bg-white p-2.5 rounded-xl border border-slate-200 text-xs space-y-1">
                  <div className="flex justify-between text-slate-600"><span>Ark Controller:</span><span className="font-bold">${outcomes.arkCost.toLocaleString()}</span></div>
                  <div className="flex justify-between text-slate-600"><span>Chamber Pod:</span><span className="font-bold">${outcomes.chamberCost.toLocaleString()}</span></div>
                </div>
                <div className="grid grid-cols-2 sm:grid-cols-4 gap-1.5 pt-1">
                  {INVESTMENT_PRESETS.map((p, i) => (
                    <button
                      key={i} onClick={() => setSetupInvestment(p.value)}
                      className={`p-2 rounded-xl text-left border text-xs ${Math.abs(setupInvestment - p.value) < 100 ? 'bg-blue-600 text-white font-bold' : 'bg-white text-slate-700 hover:bg-slate-100'}`}
                    >
                      <div className="font-mono">${p.value.toLocaleString()}</div>
                      <div className="text-[10px] truncate opacity-80">{p.label}</div>
                    </button>
                  ))}
                </div>
              </div>

              {/* Dial 2 */}
              <div className="bg-slate-50 rounded-2xl p-5 border border-slate-200 space-y-4">
                <div className="flex justify-between items-center">
                  <div>
                    <span className="text-xs font-bold text-emerald-600 uppercase">Dial 2</span>
                    <h4 className="text-sm font-bold text-slate-900">Monthly Block Capacity</h4>
                  </div>
                  <div className="text-xl font-black text-emerald-800 font-mono">{monthlyBlocks} <span className="text-xs font-normal text-slate-500">blks/mo</span></div>
                </div>
                <input
                  type="range" min="20" max="300" step="10"
                  value={monthlyBlocks} onChange={(e) => setMonthlyBlocks(Number(e.target.value))}
                  className="w-full h-2.5 bg-slate-200 rounded-lg cursor-pointer accent-emerald-600"
                />
                <div className="flex justify-between text-xs font-mono text-slate-500">
                  <span>20 blks</span><span>100 blks</span><span>200 blks</span><span>300 blks</span>
                </div>
                <div className="grid grid-cols-2 gap-2 text-xs">
                  <div className="bg-white p-2.5 rounded-xl border border-slate-200">
                    <span className="text-slate-500 block text-[11px]">Monthly Fresh Yield:</span>
                    <span className="font-bold text-slate-900 text-sm font-mono">{outcomes.monthlyYieldKg.toFixed(0)} kg / mo</span>
                  </div>
                  <div className="bg-white p-2.5 rounded-xl border border-slate-200">
                    <span className="text-slate-500 block text-[11px]">Gross Monthly Inflow:</span>
                    <span className="font-bold text-emerald-700 text-sm font-mono">${outcomes.grossMonthlySales.toLocaleString()}</span>
                  </div>
                </div>
                <div className="grid grid-cols-3 gap-1.5 pt-1">
                  {CAPACITY_PRESETS.map((p, i) => (
                    <button
                      key={i} onClick={() => setMonthlyBlocks(p.blocks)}
                      className={`p-2 rounded-xl text-left border text-xs ${monthlyBlocks === p.blocks ? 'bg-emerald-600 text-white font-bold' : 'bg-white text-slate-700 hover:bg-slate-100'}`}
                    >
                      <div className="font-mono">{p.blocks} Blocks</div>
                      <div className="text-[10px] truncate opacity-80">{p.label}</div>
                    </button>
                  ))}
                </div>
              </div>
            </div>
          </section>

          {/* Outcome Cards */}
          <section className="space-y-4">
            <div className="flex justify-between items-center">
              <div>
                <h3 className="text-lg font-bold text-slate-900">Feasibility Outcomes & Returns</h3>
                <p className="text-xs text-slate-500">Calculated directly from Fotra unit economics</p>
              </div>
              <div className="flex bg-slate-100 p-1 rounded-xl text-xs font-semibold">
                <button onClick={() => setOutcomeTab('timeline')} className={`px-3 py-1 rounded-lg ${outcomeTab === 'timeline' ? 'bg-white text-emerald-800 shadow-sm font-bold' : 'text-slate-600'}`}>Timeline</button>
                <button onClick={() => setOutcomeTab('table')} className={`px-3 py-1 rounded-lg ${outcomeTab === 'table' ? 'bg-white text-emerald-800 shadow-sm font-bold' : 'text-slate-600'}`}>12-Month Table</button>
              </div>
            </div>

            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
              <div className="bg-white rounded-2xl p-5 border border-slate-200 shadow-sm space-y-1">
                <span className="text-slate-500 text-xs font-semibold uppercase block">Monthly Net Income</span>
                <div className="text-3xl font-black text-slate-900 font-mono">${outcomes.netMonthlyOperatingProfit.toLocaleString()}<span className="text-xs font-normal text-slate-500"> / mo</span></div>
                <div className="text-xs text-emerald-700 font-bold pt-1 border-t border-slate-100">60.9% Cash Margin ($19.50/blk)</div>
              </div>

              <div className="bg-white rounded-2xl p-5 border border-slate-200 shadow-sm space-y-1">
                <span className="text-slate-500 text-xs font-semibold uppercase block">Payback Timeline</span>
                <div className="text-3xl font-black text-slate-900 font-mono">{outcomes.paybackMonths}<span className="text-sm font-normal text-slate-500 ml-1">Months</span></div>
                <div className="text-xs text-amber-800 font-bold pt-1 border-t border-slate-100">Month {outcomes.breakevenMonth} Breakeven</div>
              </div>

              <div className="bg-white rounded-2xl p-5 border border-slate-200 shadow-sm space-y-1">
                <span className="text-slate-500 text-xs font-semibold uppercase block">12-Month Net Cash</span>
                <div className="text-3xl font-black text-emerald-800 font-mono">${outcomes.year1NetCash.toLocaleString()}</div>
                <div className="text-xs text-teal-800 font-bold pt-1 border-t border-slate-100">{outcomes.year1Roi.toFixed(0)}% ROI on CapEx</div>
              </div>

              <div className="bg-white rounded-2xl p-5 border border-slate-200 shadow-sm space-y-1">
                <span className="text-slate-500 text-xs font-semibold uppercase block">Annual Net Profit</span>
                <div className="text-3xl font-black text-slate-900 font-mono">${outcomes.annualNetProfit.toLocaleString()}<span className="text-xs font-normal text-slate-500"> / yr</span></div>
                <div className="text-xs text-slate-600 font-medium pt-1 border-t border-slate-100">52-Week Harvest Run-Rate</div>
              </div>
            </div>

            {outcomeTab === 'timeline' && (
              <div className="bg-white rounded-3xl p-6 border border-slate-200 shadow-sm space-y-3">
                <div className="flex justify-between items-center pb-2 border-b border-slate-100">
                  <h4 className="text-sm font-bold text-slate-900">12-Month Cumulative Cash Recovery</h4>
                  <span className="text-xs font-bold text-emerald-800 bg-emerald-50 px-2.5 py-0.5 rounded-full border border-emerald-200">Breakeven: Month {outcomes.breakevenMonth}</span>
                </div>
                <div className="w-full h-48">
                  <svg className="w-full h-full" viewBox="0 0 600 160" preserveAspectRatio="none">
                    <line x1="40" y1="130" x2="580" y2="130" stroke="#cbd5e1" strokeWidth="1.5" strokeDasharray="3,3" />
                    <text x="35" y="133" fill="#64748b" fontSize="9" textAnchor="end" fontFamily="monospace">$0</text>
                    {outcomes.monthlySeries.map((m, idx) => {
                      const x = 55 + idx * 44;
                      const barHeight = Math.max(4, (m.netCash / outcomes.netMonthlyOperatingProfit) * 24);
                      return (
                        <g key={idx}>
                          <rect x={x - 6} y={130 - barHeight} width="12" height={barHeight} rx="2" fill={m.cumCash >= 0 ? '#a7f3d0' : '#fecdd3'} />
                          <text x={x} y="146" fill="#64748b" fontSize="9" textAnchor="middle" fontFamily="monospace">{m.shortName}</text>
                        </g>
                      );
                    })}
                  </svg>
                </div>
              </div>
            )}

            {outcomeTab === 'table' && (
              <div className="bg-white rounded-3xl p-5 border border-slate-200 shadow-sm overflow-x-auto">
                <table className="w-full text-left text-xs">
                  <thead className="bg-slate-50 text-slate-700 font-semibold border-b border-slate-200">
                    <tr>
                      <th className="py-2.5 px-3">Metric</th>
                      {outcomes.monthlySeries.map(m => <th key={m.month} className="py-2.5 px-2 text-center font-mono">{m.shortName}</th>)}
                    </tr>
                  </thead>
                  <tbody className="divide-y divide-slate-100 font-medium">
                    <tr>
                      <td className="py-2 px-3 font-semibold">Gross Sales ($)</td>
                      {outcomes.monthlySeries.map(m => <td key={m.month} className="py-2 px-2 text-center font-mono text-emerald-700">${m.inflow.toLocaleString()}</td>)}
                    </tr>
                    <tr>
                      <td className="py-2 px-3 font-semibold">Substrate OpEx ($)</td>
                      {outcomes.monthlySeries.map(m => <td key={m.month} className="py-2 px-2 text-center font-mono text-rose-700">(${Math.abs(m.substrateExpense).toLocaleString()})</td>)}
                    </tr>
                    <tr className="bg-emerald-50/60 font-bold">
                      <td className="py-2 px-3 font-bold text-emerald-950">Net Cash ($)</td>
                      {outcomes.monthlySeries.map(m => <td key={m.month} className={`py-2 px-2 text-center font-mono ${m.netCash >= 0 ? 'text-emerald-700' : 'text-rose-700'}`}>{m.netCash >= 0 ? `$${m.netCash.toLocaleString()}` : `($${Math.abs(m.netCash).toLocaleString()})`}</td>)}
                    </tr>
                    <tr className="bg-teal-50/60 font-extrabold">
                      <td className="py-2 px-3 font-extrabold text-teal-950">Cumulative ($)</td>
                      {outcomes.monthlySeries.map(m => <td key={m.month} className={`py-2 px-2 text-center font-mono ${m.cumCash >= 0 ? 'text-teal-800' : 'text-rose-700'}`}>{m.cumCash >= 0 ? `$${m.cumCash.toLocaleString()}` : `($${Math.abs(m.cumCash).toLocaleString()})`}</td>)}
                    </tr>
                  </tbody>
                </table>
              </div>
            )}
          </section>
        </div>
      );
    }

    ReactDOM.createRoot(document.getElementById('root')).render(<App />);
  </script>
</body>
</html>
